## Deployments #####################################################
## Run and show history deployments
---------------------------------------------------------------------
microk8s kubectl apply -f tican-eureka-deploy.yml
kubectl rollout history deployment tican-eureka-service

## Change Change-Couse to prefer informations
kubectl annotate deployment tican-eureka-service kubernetes.io/change-cause="start versão 2.0.0"

## Change Change-Cause to default
microk8s kubectl apply -f tican-eureka-deploy.yml --record

## Revert to old versions (from one number history)
kubectl rollout undo deployment tican-eureka-service --to-revision=1
